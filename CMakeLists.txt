cmake_minimum_required(VERSION 3.16)
project(robotlang CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-Wall -Wextra)
endif()

add_library(robotlang src/ast.cpp src/parser.cpp src/ast_semantic.cpp src/semantic.cpp src/simulator.cpp src/optimizer.cpp src/codegen.cpp src/scanner.cpp src/token.cpp)
target_include_directories(robotlang PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)

add_executable(robotc src/main.cpp)
target_link_libraries(robotc PRIVATE robotlang)

include(CTest)
if(BUILD_TESTING)
    add_executable(extension_tests tests/extension_tests.cpp)
    target_link_libraries(extension_tests PRIVATE robotlang)
    add_test(NAME extension_tests COMMAND extension_tests)
    add_test(NAME print_ast COMMAND robotc ${CMAKE_CURRENT_SOURCE_DIR}/examples/square.robot --ast)
    set_tests_properties(print_ast PROPERTIES PASS_REGULAR_EXPRESSION "Repeat 4")
    add_test(NAME generate_assembly COMMAND robotc ${CMAKE_CURRENT_SOURCE_DIR}/examples/optimization.robot --emit ${CMAKE_CURRENT_BINARY_DIR}/codegen_test.s)
    add_test(NAME generate_optimized_assembly COMMAND robotc ${CMAKE_CURRENT_SOURCE_DIR}/examples/optimization.robot --optimize --emit ${CMAKE_CURRENT_BINARY_DIR}/codegen_optimized_test.s)
    add_test(NAME generate_patrol_assembly COMMAND robotc ${CMAKE_CURRENT_SOURCE_DIR}/examples/patrol.robot --emit ${CMAKE_CURRENT_BINARY_DIR}/codegen_patrol_test.s)
    add_test(NAME generate_repeat_assembly COMMAND robotc ${CMAKE_CURRENT_SOURCE_DIR}/examples/square.robot --emit ${CMAKE_CURRENT_BINARY_DIR}/codegen_repeat_test.s)
    find_program(ROBOTLANG_GCC gcc)
    if(ROBOTLANG_GCC)
        add_test(NAME assemble_assembly COMMAND ${ROBOTLANG_GCC} -c ${CMAKE_CURRENT_BINARY_DIR}/codegen_test.s -o ${CMAKE_CURRENT_BINARY_DIR}/codegen_test.o)
        add_test(NAME assemble_optimized_assembly COMMAND ${ROBOTLANG_GCC} -c ${CMAKE_CURRENT_BINARY_DIR}/codegen_optimized_test.s -o ${CMAKE_CURRENT_BINARY_DIR}/codegen_optimized_test.o)
        add_test(NAME link_patrol_assembly COMMAND ${ROBOTLANG_GCC} ${CMAKE_CURRENT_BINARY_DIR}/codegen_patrol_test.s -o ${CMAKE_CURRENT_BINARY_DIR}/codegen_patrol_test)
        add_test(NAME link_repeat_assembly COMMAND ${ROBOTLANG_GCC} ${CMAKE_CURRENT_BINARY_DIR}/codegen_repeat_test.s -o ${CMAKE_CURRENT_BINARY_DIR}/codegen_repeat_test)
        set_tests_properties(assemble_assembly PROPERTIES DEPENDS generate_assembly)
        set_tests_properties(assemble_optimized_assembly PROPERTIES DEPENDS generate_optimized_assembly)
        set_tests_properties(link_patrol_assembly PROPERTIES DEPENDS generate_patrol_assembly)
        set_tests_properties(link_repeat_assembly PROPERTIES DEPENDS generate_repeat_assembly)
    endif()
    add_test(NAME original_scanner_square
             COMMAND robotc ${CMAKE_CURRENT_SOURCE_DIR}/examples/square.robot --tokens)
endif()
